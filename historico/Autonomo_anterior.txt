
package org.firstinspires.ftc.teamcode.Autonomo;

import com.qualcomm.hardware.rev.RevHubOrientationOnRobot;
import com.qualcomm.robotcore.eventloop.opmode.Autonomous;
import com.qualcomm.robotcore.eventloop.opmode.LinearOpMode;
import com.qualcomm.robotcore.hardware.DcMotor;
import com.qualcomm.robotcore.hardware.IMU;
import com.qualcomm.robotcore.util.ElapsedTime;

import org.firstinspires.ftc.robotcore.external.navigation.AngleUnit;

@Autonomous(name = "Autonomo", group = "Teste")
public class autonomo extends LinearOpMode {

    private DcMotor MD;
    private DcMotor ME;
    private DcMotor MI;
    private DcMotor MS;
    private IMU imu;

    private final ElapsedTime cronometroAuto = new ElapsedTime();
    private String motivoFalha = "";

    private static final double TICKS_POR_REV_MOTOR = 28.0;
    private static final double REDUCAO = 20.0;
    private static final double DIAMETRO_RODA_MM = 90.0;

    private static final double TICKS_POR_MM =
            (TICKS_POR_REV_MOTOR * REDUCAO)
                    / (Math.PI * DIAMETRO_RODA_MM);

    private static final double KP_VIRADA = 0.018;
    private static final double KD_VIRADA = 0.003;

    private static final double MIN_POTENCIA_VIRADA = 0.08;
    private static final double MAX_POTENCIA_VIRADA = 0.50;
    private static final double TOLERANCIA_VIRADA = 2.0;
    private static final double TOLERANCIA_VELOCIDADE_YAW = 8.0;
    private static final double VELOCIDADE_YAW_BAIXA = 5.0;

    private static final double KP_RUMO = 0.02;
    private static final double MAX_CORRECAO_RUMO = 0.20;
    private static final double KP_ENCODER = 0.001;
    private static final double MAX_CORRECAO_ENCODER = 0.15;
    private static final double MAX_CORRECAO_TOTAL = 0.25;

    private static final double FRACAO_DESACELERACAO = 0.20;
    private static final double MIN_POTENCIA_DESLOCAMENTO = 0.16;
    private static final double MIN_POTENCIA_RODA = 0.08;

    private static final int TOLERANCIA_DISTANCIA_TICKS = 25;
    private static final int TOLERANCIA_DIFERENCA_TICKS = 65;
    private static final int LIMITE_ULTRAPASSAGEM_TICKS = 95;
    private static final double TOLERANCIA_RUMO_DESLOCAMENTO = 5.0;

    private static final double POTENCIA_INTAKE = 0.60;
    private static final double POTENCIA_LANCAMENTO = 0.70;

    private static final double TEMPO_ACELERACAO_LANCADOR = 0.80;
    private static final double TEMPO_ALIMENTACAO_POR_PECA = 0.45;
    private static final double TEMPO_ENTRE_PECAS = 0.15;

    private static final double TEMPO_LIMITE_VIRADA = 6.0;
    private static final double TEMPO_LIMITE_DISTANCIA = 10.0;
    private static final double TEMPO_MAXIMO_AUTO = 30.0;
    private static final double RESERVA_SEGURANCA = 1.0;
    private static final double TEMPO_SEM_PROGRESSO = 1.2;
    private static final int LIMITE_ENCODER_NEGATIVO = -10;

    @Override
    public void runOpMode() {

        try {
            MD = hardwareMap.get(DcMotor.class, "MD");
            ME = hardwareMap.get(DcMotor.class, "ME");
            MI = hardwareMap.get(DcMotor.class, "MI");
            MS = hardwareMap.get(DcMotor.class, "MS");
            imu = hardwareMap.get(IMU.class, "imu");

            MD.setDirection(DcMotor.Direction.REVERSE);
            ME.setDirection(DcMotor.Direction.FORWARD);

            MD.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);
            ME.setZeroPowerBehavior(DcMotor.ZeroPowerBehavior.BRAKE);

            MI.setPower(0);
            MS.setPower(0);

            RevHubOrientationOnRobot orientacao =
                    new RevHubOrientationOnRobot(
                            RevHubOrientationOnRobot.LogoFacingDirection.LEFT,
                            RevHubOrientationOnRobot.UsbFacingDirection.UP
                    );

            boolean imuInicializado = imu.initialize(
                    new IMU.Parameters(orientacao)
            );

            MD.setMode(DcMotor.RunMode.STOP_AND_RESET_ENCODER);
            ME.setMode(DcMotor.RunMode.STOP_AND_RESET_ENCODER);

            MD.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);
            ME.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);

            telemetry.addLine("Meca Tiger - teste integrado");
            telemetry.addData("IMU inicializado", imuInicializado);
            telemetry.addData("Ticks/mm", "%.4f", TICKS_POR_MM);
            telemetry.addLine(
                    "Virar -50, coletar andando 2000 mm, virar +100, lancar 3"
            );
            telemetry.update();

            waitForStart();

            if (isStopRequested()) {
                return;
            }

            if (!imuInicializado) {
                motivoFalha = "IMU nao inicializado";
                mostrarFalha("Inicializacao");
                return;
            }

            imu.resetYaw();
            cronometroAuto.reset();

            if (!virarRelativo(-50.0)) {
                mostrarFalha("Primeira virada");
                return;
            }

            double rumoDoDeslocamento = lerYaw();

            if (!Double.isFinite(rumoDoDeslocamento)) {
                motivoFalha = "Leitura IMU invalida";
                mostrarFalha("Preparacao do deslocamento");
                return;
            }

            if (!coletarDuranteDeslocamento(
                    2000.0,
                    rumoDoDeslocamento,
                    0.35)) {

                mostrarFalha("Coleta e deslocamento");
                return;
            }

            if (!virarRelativo(100.0)) {
                mostrarFalha("Segunda virada");
                return;
            }

            if (!lancarPecasPorTempo(3)) {
                mostrarFalha("Teste do lancador");
                return;
            }

            parar();
            pararMecanismos();

            telemetry.addLine("Teste integrado concluido");
            telemetry.addData(
                    "Tempo total",
                    "%.2f s",
                    cronometroAuto.seconds()
            );
            telemetry.addData("Yaw final", "%.2f", lerYaw());
            telemetry.update();

        } finally {
            parar();
            pararMecanismos();
        }
    }

    private boolean virarRelativo(double graus) {

        double rumoInicial = lerYaw();

        if (!Double.isFinite(rumoInicial)) {
            motivoFalha = "IMU invalido no inicio da virada";
            parar();
            return false;
        }

        double rumoAlvo = normalizarAngulo(rumoInicial + graus);
        double rumoAnterior = rumoInicial;

        double velocidadeYaw = 0.0;
        double tempoEstavel = -1.0;

        ElapsedTime timer = new ElapsedTime();
        ElapsedTime amostra = new ElapsedTime();
        ElapsedTime telemetriaTimer = new ElapsedTime();

        while (opModeIsActive()
                && tempoDisponivel()
                && timer.seconds() < TEMPO_LIMITE_VIRADA) {

            double rumoAtual = lerYaw();

            if (!Double.isFinite(rumoAtual)) {
                motivoFalha = "IMU invalido durante virada";
                parar();
                return false;
            }

            double dt = amostra.seconds();

            if (dt >= 0.01) {
                double velocidadeMedida =
                        normalizarAngulo(rumoAtual - rumoAnterior) / dt;

                velocidadeYaw =
                        0.65 * velocidadeYaw + 0.35 * velocidadeMedida;

                rumoAnterior = rumoAtual;
                amostra.reset();
            }

            double erro = normalizarAngulo(rumoAlvo - rumoAtual);

            if (Math.abs(erro) <= TOLERANCIA_VIRADA
                    && Math.abs(velocidadeYaw)
                    <= TOLERANCIA_VELOCIDADE_YAW) {

                if (tempoEstavel < 0.0) {
                    tempoEstavel = timer.seconds();
                }

                if (timer.seconds() - tempoEstavel >= 0.10) {
                    parar();
                    return true;
                }

            } else {
                tempoEstavel = -1.0;
            }

            double comando = limitar(
                    KP_VIRADA * erro - KD_VIRADA * velocidadeYaw,
                    -MAX_POTENCIA_VIRADA,
                    MAX_POTENCIA_VIRADA
            );

            if (Math.abs(erro) <= TOLERANCIA_VIRADA
                    && Math.abs(velocidadeYaw)
                    > TOLERANCIA_VELOCIDADE_YAW) {

                comando = -Math.copySign(
                        Math.max(
                                MIN_POTENCIA_VIRADA,
                                Math.abs(comando)
                        ),
                        velocidadeYaw
                );

            } else if (Math.abs(erro) > TOLERANCIA_VIRADA
                    && Math.abs(comando) < MIN_POTENCIA_VIRADA
                    && Math.abs(velocidadeYaw) < VELOCIDADE_YAW_BAIXA) {

                comando = Math.copySign(
                        MIN_POTENCIA_VIRADA,
                        erro
                );
            }

            MD.setPower(comando);
            ME.setPower(-comando);

            if (telemetriaTimer.seconds() >= 0.10) {

                telemetry.addData("Movimento", "VIRANDO");
                telemetry.addData("Yaw atual", "%.2f", rumoAtual);
                telemetry.addData("Yaw alvo", "%.2f", rumoAlvo);
                telemetry.addData("Erro", "%.2f", erro);
                telemetry.addData(
                        "Yaw graus/s",
                        "%.2f",
                        velocidadeYaw
                );
                telemetry.addData(
                        "Potencia virada",
                        "%.2f",
                        comando
                );
                telemetry.update();

                telemetriaTimer.reset();
            }

            idle();
        }

        parar();

        motivoFalha = tempoDisponivel()
                ? "Tempo limite da virada"
                : "Tempo AUTO esgotado ou STOP";

        return false;
    }

    private boolean andarComRumo(
            double distanciaMm,
            double rumoAlvo,
            double velocidade) {

        if (!Double.isFinite(distanciaMm)
                || !Double.isFinite(rumoAlvo)
                || !Double.isFinite(velocidade)
                || distanciaMm <= 0.0
                || velocidade <= 0.0
                || velocidade > 1.0) {

            motivoFalha = "Parametros de deslocamento invalidos";
            parar();
            return false;
        }

        int alvoTicks = Math.max(
                1,
                (int) Math.round(distanciaMm * TICKS_POR_MM)
        );

        MD.setMode(DcMotor.RunMode.STOP_AND_RESET_ENCODER);
        ME.setMode(DcMotor.RunMode.STOP_AND_RESET_ENCODER);

        MD.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);
        ME.setMode(DcMotor.RunMode.RUN_WITHOUT_ENCODER);

        ElapsedTime timer = new ElapsedTime();
        ElapsedTime semProgresso = new ElapsedTime();
        ElapsedTime telemetriaTimer = new ElapsedTime();

        double progressoAnterior = 0.0;

        while (opModeIsActive()
                && tempoDisponivel()
                && timer.seconds() < TEMPO_LIMITE_DISTANCIA) {

            int posicaoMD = MD.getCurrentPosition();
            int posicaoME = ME.getCurrentPosition();

            if (posicaoMD < LIMITE_ENCODER_NEGATIVO
                    || posicaoME < LIMITE_ENCODER_NEGATIVO) {

                motivoFalha = "Encoder negativo: verificar direcoes";
                parar();
                return false;
            }

            int ticksMD = Math.max(0, posicaoMD);
            int ticksME = Math.max(0, posicaoME);

            double mediaTicks = (ticksMD + ticksME) / 2.0;
            int diferencaTicks = ticksMD - ticksME;

            double restantes = Math.max(
                    0.0,
                    alvoTicks - mediaTicks
            );

            double rumoAtual = lerYaw();

            if (!Double.isFinite(rumoAtual)) {
                motivoFalha = "IMU invalido durante deslocamento";
                parar();
                return false;
            }

            double erroRumo = normalizarAngulo(
                    rumoAlvo - rumoAtual
            );

            if (mediaTicks >= alvoTicks - TOLERANCIA_DISTANCIA_TICKS
                    && Math.abs(diferencaTicks)
                    <= TOLERANCIA_DIFERENCA_TICKS
                    && Math.abs(erroRumo)
                    <= TOLERANCIA_RUMO_DESLOCAMENTO) {

                parar();
                return true;
            }

            if (mediaTicks > alvoTicks + LIMITE_ULTRAPASSAGEM_TICKS) {
                motivoFalha =
                        "Passou do alvo sem equilibrar encoders/rumo";
                parar();
                return false;
            }

            if (mediaTicks >= progressoAnterior + 6.0) {
                progressoAnterior = mediaTicks;
                semProgresso.reset();

            } else if (semProgresso.seconds()
                    > TEMPO_SEM_PROGRESSO) {

                motivoFalha = "Sem progresso nos encoders por 1.2 s";
                parar();
                return false;
            }

            double correcaoRumo = limitar(
                    KP_RUMO * erroRumo,
                    -MAX_CORRECAO_RUMO,
                    MAX_CORRECAO_RUMO
            );

            double correcaoEncoder = limitar(
                    KP_ENCODER * diferencaTicks,
                    -MAX_CORRECAO_ENCODER,
                    MAX_CORRECAO_ENCODER
            );

            double faixaDesaceleracao = Math.max(
                    1.0,
                    alvoTicks * FRACAO_DESACELERACAO
            );

            double fator = limitar(
                    restantes / faixaDesaceleracao,
                    0.0,
                    1.0
            );

            double potenciaBase = Math.max(
                    MIN_POTENCIA_DESLOCAMENTO,
                    velocidade * fator
            );

            double limiteCorrecao = Math.min(
                    MAX_CORRECAO_TOTAL,
                    Math.max(
                            0.0,
                            potenciaBase - MIN_POTENCIA_RODA
                    )
            );

            double correcao = limitar(
                    correcaoRumo - correcaoEncoder,
                    -limiteCorrecao,
                    limiteCorrecao
            );

            double potenciaMD = limitar(
                    potenciaBase + correcao,
                    0.0,
                    1.0
            );

            double potenciaME = limitar(
                    potenciaBase - correcao,
                    0.0,
                    1.0
            );

            MD.setPower(potenciaMD);
            ME.setPower(potenciaME);

            if (telemetriaTimer.seconds() >= 0.10) {

                telemetry.addData("Movimento", "ANDANDO");
                telemetry.addData(
                        "Ticks MD / ME",
                        "%d / %d",
                        ticksMD,
                        ticksME
                );
                telemetry.addData("Alvo ticks", alvoTicks);
                telemetry.addData(
                        "Restante mm",
                        "%.1f",
                        restantes / TICKS_POR_MM
                );
                telemetry.addData(
                        "Diferenca ticks",
                        diferencaTicks
                );
                telemetry.addData(
                        "Erro rumo",
                        "%.2f",
                        erroRumo
                );
                telemetry.addData(
                        "Correcao total",
                        "%.3f",
                        correcao
                );
                telemetry.addData(
                        "Potencia MD / ME",
                        "%.2f / %.2f",
                        potenciaMD,
                        potenciaME
                );
                telemetry.addData(
                        "Tempo deslocamento",
                        "%.2f",
                        timer.seconds()
                );
                telemetry.update();

                telemetriaTimer.reset();
            }

            idle();
        }

        parar();

        motivoFalha = tempoDisponivel()
                ? "Tempo limite do deslocamento"
                : "Tempo AUTO esgotado ou STOP";

        return false;
    }

    private boolean coletarDuranteDeslocamento(
            double distanciaMm,
            double rumoAlvo,
            double velocidade) {

        MI.setPower(POTENCIA_INTAKE);

        try {
            return andarComRumo(
                    distanciaMm,
                    rumoAlvo,
                    velocidade
            );

        } finally {
            MI.setPower(0);
        }
    }

    private boolean lancarPecasPorTempo(int quantidade) {

        if (quantidade <= 0) {
            motivoFalha = "Quantidade invalida";
            return false;
        }

        double tempoNecessario =
                TEMPO_ACELERACAO_LANCADOR
                + quantidade * TEMPO_ALIMENTACAO_POR_PECA
                + (quantidade - 1) * TEMPO_ENTRE_PECAS;

        if (!tempoDisponivel()
                || cronometroAuto.seconds() + tempoNecessario
                >= TEMPO_MAXIMO_AUTO - RESERVA_SEGURANCA) {

            motivoFalha =
                    "Sem tempo para teste completo do lancamento";
            return false;
        }

        try {
            MS.setPower(POTENCIA_LANCAMENTO);

            if (!esperarAtivo(TEMPO_ACELERACAO_LANCADOR)) {
                motivoFalha = "Lancador interrompido ao acelerar";
                return false;
            }

            for (int i = 0; i < quantidade; i++) {

                MI.setPower(POTENCIA_INTAKE);

                if (!esperarAtivo(TEMPO_ALIMENTACAO_POR_PECA)) {
                    motivoFalha =
                            "Alimentacao do lancador interrompida";
                    return false;
                }

                MI.setPower(0);

                telemetry.addData(
                        "Tentativa de lancamento",
                        "%d / %d",
                        i + 1,
                        quantidade
                );
                telemetry.update();

                if (i < quantidade - 1
                        && !esperarAtivo(TEMPO_ENTRE_PECAS)) {

                    motivoFalha =
                            "Intervalo de lancamento interrompido";
                    return false;
                }
            }

            return true;

        } finally {
            pararMecanismos();
        }
    }

    private boolean esperarAtivo(double segundos) {

        ElapsedTime timer = new ElapsedTime();

        while (opModeIsActive()
                && tempoDisponivel()
                && timer.seconds() < segundos) {

            idle();
        }

        return opModeIsActive() && tempoDisponivel();
    }

    private double lerYaw() {

        try {
            return imu.getRobotYawPitchRollAngles()
                    .getYaw(AngleUnit.DEGREES);

        } catch (RuntimeException e) {
            return Double.NaN;
        }
    }

    private boolean tempoDisponivel() {
        return opModeIsActive()
                && cronometroAuto.seconds()
                < TEMPO_MAXIMO_AUTO - RESERVA_SEGURANCA;
    }

    private double normalizarAngulo(double graus) {

        while (graus > 180.0) {
            graus -= 360.0;
        }

        while (graus < -180.0) {
            graus += 360.0;
        }

        return graus;
    }

    private double limitar(
            double valor,
            double minimo,
            double maximo) {

        return Math.max(
                minimo,
                Math.min(maximo, valor)
        );
    }

    private void mostrarFalha(String etapa) {

        parar();
        pararMecanismos();

        telemetry.addData(
                "Autonomo interrompido",
                etapa
        );
        telemetry.addData(
                "Motivo",
                motivoFalha
        );
        telemetry.addData(
                "Tempo AUTO",
                "%.2f s",
                cronometroAuto.seconds()
        );
        telemetry.update();
    }

    private void parar() {

        if (MD != null) {
            MD.setPower(0);
        }

        if (ME != null) {
            ME.setPower(0);
        }
    }

    private void pararMecanismos() {

        if (MI != null) {
            MI.setPower(0);
        }

        if (MS != null) {
            MS.setPower(0);
        }
    }
}
